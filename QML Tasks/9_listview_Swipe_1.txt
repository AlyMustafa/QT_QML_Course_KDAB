import QtQuick 2.0
import QtQuick.Window 2.0

Window {
  visible: true
  width: 400
  height: 500
  title: "Swipe ListView"

  Rectangle {
    anchors.fill: parent
    gradient: Gradient {
      GradientStop {
        position: 0.0
        color: "#F9B2D7"
      }
      GradientStop {
        position: 1.0
        color: "#CFECF3"
      }
    }

    ListModel {
      id: myModel

      ListElement {
        name: "Ali"
      }
      ListElement {
        name: "Ahmed"
      }
      ListElement {
        name: "Mohamed"
      }
      ListElement {
        name: "Omar"
      }
      ListElement {
        name: "Youssef"
      }
      ListElement {
        name: "Mahmoud"
      }
      ListElement {
        name: "Salah"
      }
    }

    ListView {
      anchors.fill: parent
      model: myModel
      spacing: 2
      clip: true

      delegate: Item {
        id: delegateRoot

        width: ListView.view.width
        height: 70

        // Background revealed during swipe
        Rectangle {
          anchors.fill: parent
          color: "#d32f2f"

          Text {
            anchors.right: parent.right
            anchors.verticalCenter: parent.verticalCenter
            anchors.margins: 40
            text: "DELETE"
            color: "white"
            font.bold: true
          }

          MouseArea {
            anchors.fill: parent
            onClicked: {
              myModel.remove(index)
            }
          }
        }

        // Foreground item
        Rectangle {
          id: foreground

          width: parent.width
          height: parent.height
          color: "#111844"
          border.color: "#4B5694"

          Text {
            id: txt
            anchors.centerIn: parent
            text: name
            font.pixelSize: 22
            color: "white"
          }

          Behavior on x {
            NumberAnimation {
              duration: 100
            }
          }

          MouseArea {
            id: dragArea

            anchors.fill: parent

            drag.target: foreground
            drag.axis: Drag.XAxis

            drag.minimumX: -120
            drag.maximumX: 0

            onReleased: {
              if (foreground.x < -60)
                foreground.x = -120
              else
                foreground.x = 0
            }

            onDoubleClicked: {
              foreground.x = 0
            }
          }
        }
      }
    }
  }
}
